# 체감온도(Apparent Temperature) 계산 함수 정의
def calculate_at(temp, humid, wind):
    # 수증기압(e) 계산
    e = (humid / 100) * 6.105 * np.exp(17.27 * temp / (237.7 + temp))
    # 체감온도 공식 (여름철 풍속 영향 반영)
    return 1.04 * temp + 0.2 * e - 0.65 * wind - 2.7
